import agenticaiframework as aaf
from agenticaiframework.tools import ScrapeWebsiteTool

aaf.configure(provider="openai", model="gpt-4o-mini",
              guardrails="standard")

agent = aaf.Agent.quick("Assistant", role="researcher")
agent.add_tool(ScrapeWebsiteTool())

out = agent.invoke("Summarise the release notes at https://example.com/changelog")

print(out.response)
print(out.status, out.token_usage, out.trace_id)   # observable by default
